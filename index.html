'▲' : '▼'; } } }); }); // Carregar conteúdo ao clicar nos subitens document.querySelectorAll('[data-conteudo]').forEach(item => { item.addEventListener('click', function() { const chave = this.dataset.conteudo; const dados = bancoDeDados[chave]; if (dados) { document.getElementById('titulo-sistema').textContent = dados.titulo; document.getElementById('texto-sistema').textContent = dados.texto; } }); }); // Botão Início document.getElementById('btn-topo').addEventListener('click', function() { window.scrollTo({ top: 0, behavior: 'smooth' }); });
